```
using System;
using UnityEngine;

public static class GameState
{
    // Флаг, который говорит "игрок сейчас в меню/ивенте, ходить нельзя"
    public static bool IsPlayerFrozen { get; private set; }

    // События, на которые смогут подписаться другие скрипты
    public static event Action OnPlayerFrozen;
    public static event Action OnPlayerReleased;

    public static void FreezePlayer()
    {
        IsPlayerFrozen = true;
        Cursor.lockState = CursorLockMode.None; // Показываем мышку
        Cursor.visible = true;
        OnPlayerFrozen?.Invoke();
    }

    public static void ReleasePlayer()
    {
        IsPlayerFrozen = false;
        Cursor.lockState = CursorLockMode.Locked; // Прячем мышку
        Cursor.visible = false;
        OnPlayerReleased?.Invoke();
    }
}
```

### 2. Система ресурсов (С предупреждениями)

Создай скрипт ResourceManager.cs. Добавление порога в 20% — гениальный геймдизайнерский ход. Это создаст ту самую панику, когда экран краснеет, а сердце колотится.

expand_more

```
using System;
using System.Collections.Generic;
using UnityEngine;

public enum ResourceType { Health, Sanity, Hunger, Money }

public class ResourceManager : MonoBehaviour
{
    public static ResourceManager Instance { get; private set; }

    private Dictionary<ResourceType, float> _values = new Dictionary<ResourceType, float>();

    // События для UI и звуков
    public event Action<ResourceType, float> OnResourceChanged;
    public event Action<ResourceType> OnResourceCritical; // < 20%
    public event Action<ResourceType> OnResourceDepleted; // <= 0%

    private void Awake()
    {
        Instance = this;
        // Инициализируем стартовые значения (по 100)
        foreach (ResourceType type in Enum.GetValues(typeof(ResourceType)))
        {
            _values[type] = 100f;
        }
    }

    public void Modify(ResourceType type, float amount)
    {
        _values[type] = Mathf.Clamp(_values[type] + amount, 0f, 100f);
        
        OnResourceChanged?.Invoke(type, _values[type]); // Обновляем UI

        if (_values[type] <= 20f && _values[type] > 0f)
        {
            OnResourceCritical?.Invoke(type); // Включаем звук сердцебиения/красный экран
        }

        if (_values[type] <= 0f)
        {
            OnResourceDepleted?.Invoke(type); // Запускаем экран Game Over
        }
    }

    // Метод для чтения значений (например, для UI)
    public float GetValue(ResourceType type) => _values[type];
}
```

```
using UnityEngine;

public class EventAnchor : MonoBehaviour, IInteractable
{
    public EventLocation locationType;
    
    [SerializeField] private ParticleSystem localVFX; 
    [SerializeField] private AudioSource localAudio;

    private ActiveEvent _currentEvent; // Ссылка на текущее активное событие

    public void TriggerEvent(EventData data)
    {
        if (localAudio != null && data.alertSound != null)
        {
            localAudio.clip = data.alertSound;
            localAudio.Play();
        }
        
        if (localVFX != null) localVFX.Play();

        // Вешаем компонент-таймер прямо на этот объект!
        _currentEvent = gameObject.AddComponent<ActiveEvent>();
        _currentEvent.Begin(data, this);
    }

    public void Interact()
    {
        if (_currentEvent != null)
        {
            // Игрок успел подойти и нажать E!
            _currentEvent.Success();
            StopEffects();
            
            // Здесь вызываем UI с выбором (UIManager.Instance.ShowEventPanel(data))
        }
    }

    public string GetPromptText() => "Взаимодействовать";

    public void StopEffects()
    {
        if (localAudio != null) localAudio.Stop();
        if (localVFX != null) localVFX.Stop();
    }
}
```

```
using UnityEngine;

public class ActiveEvent : MonoBehaviour
{
    private EventData _data;
    private EventAnchor _anchor;
    private float _timer;
    private bool _isActive;

    public void Begin(EventData data, EventAnchor anchor)
    {
        _data = data;
        _anchor = anchor;
        _timer = data.timeToReact;
        _isActive = true;
    }

    private void Update()
    {
        if (!_isActive) return;

        _timer -= Time.deltaTime;
        
        if (_timer <= 0f) 
        {
            ApplyPenaltyAndEnd();
        }
    }

    private void ApplyPenaltyAndEnd()
    {
        _isActive = false;
        
        // Снимаем ресурсы за игнор
        ResourceManager.Instance.Modify(_data.penaltyType, -_data.penaltyAmount);
        
        _anchor.StopEffects();
        Destroy(this); // Удаляем этот скрипт, событие завершено
    }

    public void Success()
    {
        _isActive = false;
        Destroy(this); // Удаляем скрипт таймера, штрафа не будет
    }
}
```

### Что это дает на практике?

1. Ты создаешь на сцене дверь. Кидаешь на нее скрипт EventAnchor и настраиваешь AudioSource (звук стука).
    
2. Твой EventManager (который просто раз в минуту кидает рандом) говорит: "Дверь, держи EventData (Пришел сосед)".
    
3. Дверь сама включает звук, сама вешает на себя таймер ActiveEvent.
    
4. Если ты не успел подойти за 15 секунд — таймер снимает тебе ресурсы и сам себя удаляет. Звук стука прекращается.
    
5. Никаких спагетти-связей!
